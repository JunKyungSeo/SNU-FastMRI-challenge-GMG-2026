# SWDC (Spatially-Weighted Data Consistency) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** PromptMR+의 cascade DC를 "스칼라 dc_weight" 에서 "k-space 위치별 학습 가중치 지도(SWDC, SDUM §3.4)" 로 바꾼다. **CLI 레버 없이 기본 동작**(사용자 지시 2026-08-07 — EMA와 동일 방침). A/B는 커밋 전 기존 실측(shell32)과 비교한다.

**Architecture:** SDUM의 `dc_weight_map`을 복붙할 수 없다 — SDUM은 DC를 k-space에서 직접 하지만, PromptMR+는 이미지 도메인에서 `soft_dc = (ffx − img_zf) × η` (η=스칼라)로 한다. 공간가변 가중은 **A^H(sens_reduce) 앞, k-space에서** 곱해야 주파수별 의미가 생기므로, cascade가 k-space 잔차 `M·A·x − k0`를 직접 만들도록 DC 경로를 재구성한다: `soft_dc = A^H(W ⊙ (M·A·x − k0))`. 이를 위해 `masked_kspace`(k0)를 cascade forward에 새로 전달한다. W=1(초기값)이면 선형성에 의해 기존 η=1과 수학적으로 동일 → 등가성 테스트로 강제한다. `swdc=False`는 **테스트 전용 내부 kwarg**로만 남긴다(등가성 비교 기준) — CLI·config에는 노출하지 않는다.

**Tech Stack:** PyTorch 2.3.1, pytest (CPU, float64 tiny net — 기존 `test_checkpoint_equivalence.py` 패턴 재사용)

# test_checkpoint_equivalence.py의 TINY와 같은 값(자체 보유 — 테스트 간 import 안 함)
TINY = dict(
    num_cascades=2,
    n_feat0=6,
    feature_dim=[8, 10, 12],
    prompt_dim=[4, 6, 8],
    sens_n_feat0=6,
    sens_feature_dim=[8, 10, 12],
    sens_prompt_dim=[4, 6, 8],
    prompt_size=[8, 4, 2],
    n_history=2,
    n_buffer=4,
    adaptive_input=True,

def _make_net(swdc, seed=0):
    # dc_weight(ones)·dc_weight_map(ones) 둘 다 RNG를 안 쓰므로, 같은 seed면
    # swdc on/off 넷의 나머지 가중치가 전부 같다 → 초기값 등가성 비교가 성립한다.
    torch.manual_seed(seed)
    return PromptMR(**TINY, swdc=swdc).double()


def _make_batch(seed=1):
    g = torch.Generator().manual_seed(seed)
    kspace = torch.randn(1, N_COILS, H, W, 2, generator=g, dtype=torch.float64)
    mask1d = torch.zeros(W, dtype=torch.float64)
    mask1d[::3] = 1.0
    mask1d[W // 2 - 2: W // 2 + 2] = 1.0
    mask = mask1d.view(1, 1, 1, W, 1).expand(1, 1, H, W, 1).contiguous()
    return kspace * mask, mask


def _forward(net, kspace, mask, use_checkpoint=False):
    return net(kspace, mask, num_low_frequencies=None, mask_type=("cartesian",),
               use_checkpoint=use_checkpoint, compute_sens_per_coil=True)


def test_swdc_is_default():
    """레버가 없으므로 기본 생성 = SWDC. config/CLI 배선 없이 모든 신규 학습에 실려야 한다."""
    torch.manual_seed(0)
    net = PromptMR(**TINY)
    for blk in net.cascades:
        assert blk.dc_weight_map.shape == (1, 1, 640, 480, 1)
        assert not hasattr(blk, "dc_weight")


def test_swdc_off_keeps_scalar_dc():
    """swdc=False는 테스트 전용 비교 기준 — 기존 구조(스칼라)가 그대로여야 한다."""
    net = _make_net(swdc=False)
    for blk in net.cascades:
        assert hasattr(blk, "dc_weight") and blk.dc_weight.shape == (1,)
        assert not hasattr(blk, "dc_weight_map")


def test_swdc_init_equals_scalar_baseline():
    """W=1 초기값에서 swdc 경로는 스칼라 η=1 경로와 수학적으로 같아야 한다."""
    kspace, mask = _make_batch()
    with torch.no_grad():
        out_off = _forward(_make_net(swdc=False).eval(), kspace, mask)
        out_on = _forward(_make_net(swdc=True).eval(), kspace, mask)
    torch.testing.assert_close(out_on, out_off, rtol=1e-6, atol=1e-8)


def test_swdc_map_receives_grad():
    net = _make_net(swdc=True).train()
    kspace, mask = _make_batch()
    _forward(net, kspace, mask).square().mean().backward()
    for i, blk in enumerate(net.cascades):
        g = blk.dc_weight_map.grad
        assert g is not None, f"cascade {i}: dc_weight_map grad 없음"
        # 샘플된 열에만 잔차가 있으므로 grad는 그 열에서 0이 아니어야 한다
        assert g.abs().sum() > 0, f"cascade {i}: dc_weight_map grad 전부 0"

class PromptMRBlock(nn.Module):
    def __init__(self, model: nn.Module, num_adj_slices: int = 1, swdc: bool = True):

        super().__init__()
        self.model = model
        self.num_adj_slices = num_adj_slices   # DC(sens_expand/reduce)가 슬라이스 그룹을 알아야 한다
        self.swdc = swdc
        if swdc:
            # SWDC(SDUM §3.4): 스칼라 η 대신 k-space 위치별 학습 가중치 지도. (640, 480) =
            # readout-crop 없는 H 최대 × 학습셋 W 최대 — 실제 (H, W)로 중앙 crop해 쓴다.
            # ones 초기화라 학습 시작 시점엔 η=1 스칼라와 수학적으로 동일(테스트로 강제).
            self.dc_weight_map = nn.Parameter(torch.ones(1, 1, 640, 480, 1))
        else:
            # 테스트 전용 비교 기준(등가성). CLI/config에서 이 분기로 못 들어온다.
            self.dc_weight = nn.Parameter(torch.ones(1))

    def forward(
        self,
        current_img: torch.Tensor,
        img_zf: torch.Tensor,
        latent: torch.Tensor,
        mask: torch.Tensor,
        sens_maps: torch.Tensor,
        history_feat: Optional[Tuple[torch.Tensor, ...]] = None,
        masked_kspace: Optional[torch.Tensor] = None,
    ):
        zero = torch.zeros(1, 1, 1, 1, 1).to(current_img)
        current_kspace = fastmri.sens_expand(current_img, sens_maps, self.num_adj_slices)
        ffx = fastmri.sens_reduce(torch.where(mask != 0, current_kspace, zero), sens_maps,
                                  self.num_adj_slices) # mask != 0 for silence warning
        if self.model.n_buffer > 0:
            # adaptive input. buffer: A^H*A*x_i, s_i, x0, A^H*A*x_i-x0
            buffer = torch.cat([ffx, latent, img_zf] + [ffx-img_zf]*(self.model.n_buffer-3), dim=1)
        else:
            buffer = None

        if self.swdc:
            # soft_dc = A^H(W ⊙ (M·A·x − k0)). W는 A^H **앞**(k-space)에서 곱해야 주파수별
            # 의미가 생긴다 — (ffx − img_zf)에 곱하면 이미지 도메인 가중이라 다른 물건이다.
            # k0(masked_kspace)는 mask 밖이 0이라 where 하나로 잔차가 닫힌다.
            assert masked_kspace is not None, "swdc=True면 masked_kspace를 넘겨야 한다"
            _, _, h, w, _ = current_kspace.shape
            assert h <= 640 and w <= 480, f"dc_weight_map(640x480)보다 큰 k-space {h}x{w}"
            wmap = self.dc_weight_map[:, :, (640 - h) // 2: (640 - h) // 2 + h,
                                      (480 - w) // 2: (480 - w) // 2 + w, :]
            k_res = torch.where(mask != 0, current_kspace - masked_kspace, zero)
            soft_dc = fastmri.sens_reduce(wmap * k_res, sens_maps, self.num_adj_slices)
        else:
            soft_dc = (ffx - img_zf) * self.dc_weight
        model_term, latent, history_feat = self.model(current_img, history_feat, buffer)
        img_pred = current_img - soft_dc - model_term
        return img_pred, latent, history_feat
```

def test_swdc_checkpoint_equivalence():
    """swdc 경로에서 checkpoint on/off가 출력·모든 grad에서 일치해야 한다(float64)."""
    kspace, mask = _make_batch()

    def run(use_checkpoint):
        net = _make_net(swdc=True)
        net.train()
        net.zero_grad(set_to_none=True)
        out = _forward(net, kspace, mask, use_checkpoint=use_checkpoint)
        out.square().mean().backward()
        grads = {n: p.grad.detach().clone() for n, p in net.named_parameters()
                 if p.grad is not None}
        return out.detach(), grads

    out_ckp, grad_ckp = run(True)
    out_raw, grad_raw = run(False)
    torch.testing.assert_close(out_ckp, out_raw, rtol=1e-9, atol=1e-9)
    assert grad_ckp.keys() == grad_raw.keys()
    assert any(n.endswith("dc_weight_map") for n in grad_ckp), "map grad가 안 잡혔다"
    for name in grad_ckp:
        torch.testing.assert_close(
            grad_ckp[name], grad_raw[name], rtol=1e-9, atol=1e-9,
            msg=lambda m, n=name: f"param '{n}' grad가 checkpoint 여부에 따라 다름:\n{m}")
```

## 계획에서 뺀 것 (ponytail — 이유와 복귀 조건)

- **CLI 레버(`--swdc`)**: 사용자 지시로 제외(2026-08-07) — EMA처럼 기본 동작. A/B OFF arm은 shell22 전례처럼 기존 실측(shell32)과 비교하고, 지면 revert로 되돌린다.
- **mask 종류별 지도 분기** (SDUM `mask_specific_dc_weight_map`): 이 대회 mask는 uniform 1종 — 지도 1장이면 된다. mask family를 추가하면 그때 인덱싱 복귀.
- **replicate-pad** (SDUM의 지도보다 큰 입력 대응): 데이터 상한이 640×480으로 고정 — assert로 시끄럽게 죽는 쪽을 택했다. 데이터가 바뀌면 pad 복귀.
- **옛 ckpt warm-start 변환** (스칼라→지도 초기화): A/B는 처음부터 학습으로 판정하고, 이기면 본학습도 처음부터(shell31 3단계) — 변환 코드는 그때도 불필요.
- **per-cascade 공유 지도**: 현행 스칼라도 cascade별이므로 지도도 cascade별(10장, +3.1M 파라미터·+12MB — 무시 가능). 공유는 절약 이유가 없다.
- **EMA·MoE·test_part 수정**: 전부 state_dict/저장된 args 기반이라 자동 호환 — 손대지 않는다.
